cmake_minimum_required(VERSION 3.24)
project(SLX_ReelSync VERSION 1.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(NOT TARGET SLX_ReelSync_Core)
  include("${CMAKE_CURRENT_SOURCE_DIR}/sdk/cmake/SiliconLogiXReelSyncConfig.cmake")
else()
  add_library(SiliconLogiX::ReelSync ALIAS SLX_ReelSync_Core)
endif()
set(SLX_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/third_party" CACHE PATH "Pinned dependencies")
option(SLX_BUILD_DESKTOP "Build PC OpenGL interface" ON)
add_library(slx-reelsync-educational STATIC examples/educational_controller.c)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/examples/educational_controller.c" SLX_EDUCATIONAL_HASH)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS examples/educational_controller.c)
target_compile_definitions(slx-reelsync-educational PRIVATE SLX_RS_EDUCATIONAL_HASH="${SLX_EDUCATIONAL_HASH}")
target_include_directories(slx-reelsync-educational PUBLIC examples sdk/include)
add_library(slx-reelsync-host STATIC desktop/session.cpp desktop/protocol.cpp desktop/remote.cpp)
if(WIN32)
  target_link_libraries(slx-reelsync-host PUBLIC winhttp)
endif()
target_include_directories(slx-reelsync-host PUBLIC desktop sdk/include "${SLX_DEPS}/include")
target_link_libraries(slx-reelsync-host PUBLIC SiliconLogiX::ReelSync slx-reelsync-educational)
add_executable(slx-reelsync-cli desktop/cli.cpp)
target_link_libraries(slx-reelsync-cli PRIVATE slx-reelsync-host)
if(SLX_BUILD_DESKTOP)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${SLX_DEPS}/glfw" "${CMAKE_BINARY_DIR}/glfw" EXCLUDE_FROM_ALL)
  find_package(OpenGL REQUIRED)
  set(IMGUI "${SLX_DEPS}/imgui")
  add_library(slx-reelsync-ui STATIC ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp
    ${IMGUI}/backends/imgui_impl_glfw.cpp ${IMGUI}/backends/imgui_impl_opengl3.cpp ${SLX_DEPS}/implot/implot.cpp ${SLX_DEPS}/implot/implot_items.cpp)
  target_include_directories(slx-reelsync-ui SYSTEM PUBLIC ${IMGUI} ${IMGUI}/backends ${SLX_DEPS}/implot ${SLX_DEPS}/cgltf ${SLX_DEPS}/stb ${SLX_DEPS}/include)
  target_link_libraries(slx-reelsync-ui PUBLIC glfw OpenGL::GL)
  add_executable(SLX-ReelSync WIN32 desktop/app.cpp desktop/renderer.cpp desktop/ui.cpp desktop/platform.cpp desktop/instrument_panel.cpp)
  if(WIN32)
    enable_language(RC)
    set(SLX_BRAND_ICON "${CMAKE_CURRENT_SOURCE_DIR}/branding/reelsync.ico")
    configure_file(desktop/windows.rc.in "${CMAKE_CURRENT_BINARY_DIR}/reelsync.rc" @ONLY)
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/reelsync.rc" PROPERTIES OBJECT_DEPENDS "${SLX_BRAND_ICON}")
    target_sources(SLX-ReelSync PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/reelsync.rc")
    add_executable(SLX-ReelSync-Launcher WIN32 desktop/launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/reelsync.rc")
    target_include_directories(SLX-ReelSync-Launcher PRIVATE sdk/include)
    target_link_libraries(SLX-ReelSync-Launcher PRIVATE user32)
    target_link_options(SLX-ReelSync-Launcher PRIVATE -municode -static-libgcc -static-libstdc++)
    add_custom_command(TARGET SLX-ReelSync-Launcher POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SLX-ReelSync-Launcher>" "${CMAKE_BINARY_DIR}/SLX-ReelSync.exe")
  endif()
  target_link_libraries(SLX-ReelSync PRIVATE slx-reelsync-host slx-reelsync-ui)
  if(WIN32)
    target_link_libraries(SLX-ReelSync PRIVATE shell32 ole32 comdlg32 gdi32)
  endif()
endif()
if(NOT TARGET SLX_ReelSync_Core)
  foreach(t slx-reelsync-cli SLX-ReelSync)
    if(TARGET ${t})
      add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SiliconLogiX::ReelSync>" "$<TARGET_FILE_DIR:${t}>")
      add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/evaluation.license" "${CMAKE_BINARY_DIR}/evaluation.license")
      if(t STREQUAL "SLX-ReelSync")
        add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/bin/slx-reelsync-link.exe" "${CMAKE_BINARY_DIR}/bin/slx-reelsync-link.exe")
        add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "${CMAKE_BINARY_DIR}/assets")
        add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/docs" "${CMAKE_BINARY_DIR}/docs")
        add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "${CMAKE_BINARY_DIR}/licenses")
        foreach(doc LICENSE SDK_LICENSE ASSET_LICENSE NOTICE THIRD_PARTY_NOTICES.md)
          add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/${doc}" "${CMAKE_BINARY_DIR}/${doc}")
        endforeach()
      endif()
    endif()
  endforeach()
endif()
if(WIN32)
  foreach(t slx-reelsync-cli SLX-ReelSync)
    if(TARGET ${t} AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_link_options(${t} PRIVATE -static-libgcc -static-libstdc++)
    endif()
  endforeach()
endif()
